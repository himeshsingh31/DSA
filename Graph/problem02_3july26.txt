#include<unordered_map>
#include<list>
#include<string>
#include<queue>
bool dfs(int n,unordered_map<int,list<int>>&adj,unordered_map<int,int>&parent,unordered_map<int,bool>&visited){

queue<int> q;


while(!q.empty()){
int x = q.front();
q.pop();

for(auto i:adj[x]){

if(visited[i]==1 && parent[i]!=x){
    return true;
}
else if(visited[i]==0){
    q.push(i);
    visited[i]=1;
parent[i] = x;
}
}

}
return false;

}

{
    // Write your code here.
