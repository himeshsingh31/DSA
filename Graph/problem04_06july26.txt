#include <bits/stdc++.h> 
#include<vector>
void topologicaldfs(vector<bool>&visited,int node,stack<int>&s,unordered_map<int,list<int>>&adj){
    visited[node] = true;

    for(auto i:adj[node]){
    if(!visited[i])
        topologicaldfs(visited,i,s,adj);
    }
s.push(node);
}

vector<int> topologicalSort(vector<vector<int>> &edges, int v, int e)  {

    unordered_map<int,list<int>>adj;

    for(int i=0;i<edges.size();i++){
        int u = edges[i][0];
        int v = edges[i][1];

        adj[u].push_back(v);
    }
    vector<bool>visited(v,false);
stack<int>s;
vector<int>ans;
    for(int i =0;i<v;i++){
        if(!visited[i]){
            topologicaldfs(visited,i,s,adj);
        }

    }

while(!s.empty()){

s.pop();
